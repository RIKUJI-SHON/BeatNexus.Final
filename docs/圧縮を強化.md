## 現状の問題点

### 1. 圧縮閾値が高すぎる
- **現在**: 300MB以上のファイルのみ圧縮
- **問題**: 50-300MBの動画が無圧縮でアップロード
- **影響**: iOS端末でのメモリ制限超過

### 2. 圧縮パラメータが緩い
- **現在の推定設定**:
  - ビットレート: 制限なしまたは高め
  - 解像度: オリジナル維持
  - コーデック: デフォルト設定

### 3. iOS最適化の欠如
- H.264コーデックの明示的指定なし
- プロファイル設定の不足
- モバイル向け最適化なし

#### 軽圧縮（高品質維持）
```bash
ffmpeg -i input.mp4 \
  -c:v libx264 -profile:v baseline -level 3.1 \
  -crf 23 -b:v 1500k -maxrate 2000k -bufsize 2000k \
  -vf "scale='min(1280,iw)':min'(720,ih)':force_original_aspect_ratio=decrease,scale=trunc(iw/2)*2:trunc(ih/2)*2" \
  -c:a aac -b:a 128k -ar 44100 \
  -movflags +faststart \
  -preset medium \
  output.mp4
```

```typescript
class ProgressiveCompressor {
  async compress(file: File, targetSize: number): Promise<Blob> {
    let compressed = await this.firstPass(file);
    
    // サイズチェック
    if (compressed.size > targetSize) {
      // 2回目の圧縮（より強力）
      compressed = await this.secondPass(compressed, targetSize);
    }
    
    return compressed;
  }
  
  private async firstPass(file: File): Promise<Blob> {
    // 初回圧縮（品質優先）
    return await compressWithProfile(file, 'light');
  }
  
  private async secondPass(blob: Blob, targetSize: number): Promise<Blob> {
    // 2回目圧縮（サイズ優先）
    const currentSize = blob.size;
    const reductionRatio = targetSize / currentSize;
    
    // ビットレートを動的調整
    const adjustedBitrate = Math.floor(1000 * reductionRatio);
    return await compressWithCustomBitrate(blob, adjustedBitrate);
  }
}
```

// 既存コードに追加・修正

import { FFmpeg } from '@ffmpeg/ffmpeg';
import { fetchFile } from '@ffmpeg/util';

// 圧縮プロファイル定義
interface CompressionProfile {
  name: string;
  videoBitrate: string;
  audioBitrate: string;
  crf: number;
  maxSize: number; // MB単位
}

const COMPRESSION_PROFILES: Record<string, CompressionProfile> = {
  light: {
    name: 'light',
    videoBitrate: '1500k',
    audioBitrate: '128k',
    crf: 23,
    maxSize: 20
  },
  medium: {
    name: 'medium',
    videoBitrate: '1200k',
    audioBitrate: '96k',
    crf: 25,
    maxSize: 25
  },
  heavy: {
    name: 'heavy',
    videoBitrate: '1000k',
    audioBitrate: '96k',
    crf: 28,
    maxSize: 30
  },
  extreme: {
    name: 'extreme',
    videoBitrate: '800k',
    audioBitrate: '64k',
    crf: 30,
    maxSize: 20
  }
};

export const useVideoProcessor = () => {
  const [isLoading, setIsLoading] = useState(false);
  const [progress, setProgress] = useState(0);
  const [currentStage, setCurrentStage] = useState<string>('');
  const [isReady, setIsReady] = useState(false);
  const ffmpegRef = useRef<FFmpeg | null>(null);

  // 圧縮プロファイル選択
  const getCompressionProfile = (fileSize: number): CompressionProfile => {
    const sizeMB = fileSize / (1024 * 1024);
    
    if (sizeMB < 30) {
      return COMPRESSION_PROFILES.light;
    } else if (sizeMB < 100) {
      return COMPRESSION_PROFILES.medium;
    } else if (sizeMB < 200) {
      return COMPRESSION_PROFILES.heavy;
    } else {
      return COMPRESSION_PROFILES.extreme;
    }
  };

  // iOS最適化されたFFmpegコマンド構築
  const buildFFmpegCommand = (profile: CompressionProfile): string[] => {
    return [
      '-i', 'input.mp4',
      // ビデオコーデック設定
      '-c:v', 'libx264',
      '-profile:v', 'baseline',  // iOS互換性最大化
      '-level', '3.1',           // iOS互換性レベル
      '-crf', profile.crf.toString(),
      '-b:v', profile.videoBitrate,
      '-maxrate', `${parseInt(profile.videoBitrate) * 1.5}k`,
      '-bufsize', `${parseInt(profile.videoBitrate) * 2}k`,
      // 解像度制限（最大720p）
      '-vf', "scale='min(1280,iw)':min'(720,ih)':force_original_aspect_ratio=decrease,scale=trunc(iw/2)*2:trunc(ih/2)*2",
      // オーディオ設定
      '-c:a', 'aac',
      '-b:a', profile.audioBitrate,
      '-ar', '44100',
      '-ac', '2',  // ステレオ
      // iOS最適化
      '-movflags', '+faststart',  // ストリーミング最適化
      '-pix_fmt', 'yuv420p',      // iOS互換ピクセルフォーマット
      // パフォーマンス設定
      '-preset', 'medium',
      '-threads', '4',
      // 出力
      'output.mp4'
    ];
  };

  const processVideo = async (file: File): Promise<File> => {
    if (!ffmpegRef.current) {
      throw new Error('FFmpeg is not initialized');
    }

    setIsLoading(true);
    setProgress(0);
    
    try {
      const ffmpeg = ffmpegRef.current;
      const profile = getCompressionProfile(file.size);
      
      console.log(`🎬 Compressing video with profile: ${profile.name}`);
      console.log(`📊 Original size: ${(file.size / 1024 / 1024).toFixed(1)}MB`);
      console.log(`🎯 Target max size: ${profile.maxSize}MB`);
      
      // Stage 1: ファイル読み込み
      setCurrentStage(t('videoProcessor.loadingFile'));
      setProgress(10);
      await ffmpeg.writeFile('input.mp4', await fetchFile(file));
      
      // Stage 2: 圧縮処理
      setCurrentStage(t('videoProcessor.compressing'));
      setProgress(20);
      
      // プログレス監視
      ffmpeg.on('progress', ({ progress: ffmpegProgress }) => {
        // 20-90%の範囲でプログレス表示
        const scaledProgress = 20 + (ffmpegProgress * 70);
        setProgress(Math.round(scaledProgress));
      });
      
      // FFmpeg実行
      const command = buildFFmpegCommand(profile);
      console.log('🔧 FFmpeg command:', command.join(' '));
      
      await ffmpeg.exec(command);
      
      // Stage 3: 出力ファイル取得
      setCurrentStage(t('videoProcessor.finalizing'));
      setProgress(95);
      
      const data = await ffmpeg.readFile('output.mp4');
      const blob = new Blob([data], { type: 'video/mp4' });
      
      // サイズチェック
      const compressedSizeMB = blob.size / (1024 * 1024);
      console.log(`✅ Compressed size: ${compressedSizeMB.toFixed(1)}MB`);
      console.log(`📉 Reduction: ${((1 - blob.size / file.size) * 100).toFixed(1)}%`);
      
      // ターゲットサイズを超えている場合の警告
      if (compressedSizeMB > profile.maxSize * 1.2) {
        console.warn(`⚠️ Compressed size exceeds target by ${((compressedSizeMB / profile.maxSize - 1) * 100).toFixed(1)}%`);
        
        // 必要に応じて2回目の圧縮（より強力な設定で）
        if (compressedSizeMB > 50 && profile.name !== 'extreme') {
          console.log('🔄 Applying second pass compression...');
          setCurrentStage(t('videoProcessor.optimizing'));
          
          // より強力な圧縮プロファイルで再圧縮
          const strongerProfile = COMPRESSION_PROFILES.extreme;
          const secondPassCommand = buildFFmpegCommand(strongerProfile);
          
          await ffmpeg.writeFile('input.mp4', data);
          await ffmpeg.exec(secondPassCommand);
          
          const secondPassData = await ffmpeg.readFile('output.mp4');
          const secondPassBlob = new Blob([secondPassData], { type: 'video/mp4' });
          
          console.log(`✅ Second pass size: ${(secondPassBlob.size / 1024 / 1024).toFixed(1)}MB`);
          
          // クリーンアップ
          await ffmpeg.deleteFile('input.mp4');
          await ffmpeg.deleteFile('output.mp4');
          
          const fileName = file.name.replace(/\.[^/.]+$/, '') + '_compressed.mp4';
          return new File([secondPassBlob], fileName, { type: 'video/mp4' });
        }
      }
      
      // クリーンアップ
      await ffmpeg.deleteFile('input.mp4');
      await ffmpeg.deleteFile('output.mp4');
      
      setProgress(100);
      
      // File オブジェクトとして返す
      const fileName = file.name.replace(/\.[^/.]+$/, '') + '_compressed.mp4';
      return new File([blob], fileName, { type: 'video/mp4' });
      
    } catch (error) {
      console.error('❌ Video processing error:', error);
      throw error;
    } finally {
      setIsLoading(false);
      setCurrentStage('');
    }
  };

  // ... 既存のコード ...

  return {
    processVideo,
    isLoading,
    progress,
    currentStage,
    isReady,
    // 新しく追加
    getCompressionProfile,
    COMPRESSION_PROFILES
  };
};

const performSubmission = async () => {
  if (!videoFile || !acceptedGuidelines || !acceptedFacePolicy || !acceptedContent) return;

    if (videoFile.size > compressionThreshold) {

      try {

        const processedResult = await processVideo(videoFile);
        
        if (processedResult instanceof File) {
          finalVideoFile = processedResult;
        } else {
          const fileName = videoFile.name.replace(/\.[^/.]+$/, '') + '_compressed.mp4';
          finalVideoFile = new File([processedResult], fileName, { type: 'video/mp4' });
        }
        
        const compressedSizeMB = finalVideoFile.size / 1024 / 1024;
        const reductionPercent = ((1 - finalVideoFile.size / videoFile.size) * 100).toFixed(1);
        
        console.log(`✅ Compression completed:`);
        console.log(`   Original: ${originalSizeMB.toFixed(1)}MB`);
        console.log(`   Compressed: ${compressedSizeMB.toFixed(1)}MB`);
        console.log(`   Reduction: ${reductionPercent}%`);
        
        // iOS向け最終チェック（50MB以下を推奨）
        if (compressedSizeMB > 50) {
          console.warn(`⚠️ File size (${compressedSizeMB.toFixed(1)}MB) may cause issues on iOS devices`);
          // 必要に応じてユーザーに警告を表示
          setSubmissionStage(t('submissionModal.warning.largefile'));
        }
        
        setSubmissionProgress(50);
        setIsCompressing(false);
        setSubmissionStage(t('submissionModal.uploading'));
      } catch (compressionError) {
        console.error('Compression error:', compressionError);
        setSubmissionError(compressionError instanceof Error ? compressionError.message : t('postPage.errors.videoProcessingFailed'));
        setIsSubmissionProcessing(false);
        setIsCompressing(false);
        return;
      }
    } else {
      // 10MB未満でも品質最適化のため軽い圧縮を推奨
      console.log(`📦 File size (${(videoFile.size / 1024 / 1024).toFixed(1)}MB) is below threshold, considering light compression...`);
      
      // オプション: 小さいファイルでも軽い圧縮を適用
      try {
        const processedResult = await processVideo(videoFile);
        if (processedResult instanceof File) {
          finalVideoFile = processedResult;
          console.log(`✅ Light compression applied: ${(finalVideoFile.size / 1024 / 1024).toFixed(1)}MB`);
        }
      } catch (error) {
        console.log('Light compression skipped, using original file');
      }
      
      setSubmissionProgress(50);
    }
    
    // ... 既存のアップロード処理 ...
  } catch (err) {
    console.error('Submission error:', err);
    setSubmissionError(err instanceof Error ? err.message : '投稿に失敗しました');
    setIsSubmissionProcessing(false);
  }
};
